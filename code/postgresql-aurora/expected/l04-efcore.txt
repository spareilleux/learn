SELECT i.name AS "Name", i.guitar_voicing AS "GuitarVoicing", i.alternate_names[1] AS "FirstAlias"
FROM ga.iconic_chords AS i
WHERE 4 = ANY (i.pitch_classes) AND i.guitar_voicing IS NOT NULL
ORDER BY i.name

SELECT i.name
FROM ga.iconic_chords AS i
WHERE EXISTS (
    SELECT 1
    FROM unnest(i.alternate_names) AS a(value)
    WHERE a.value ILIKE '%beatles%')
